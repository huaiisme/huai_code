```{code-block} shell
:linenos:
:caption: 编译obstacle_fusion
cd /home/huai/work/huai_work/XCMG_work/MultiSensorPostFusion/catkin_ws

# 1. 清理这个包的编译产物（必须）
rm -rf build/target_tracking devel/target_tracking

# 2. 进入build目录，直接make指定目标 fusion
cd build
make fusion -j$(nproc)
cd ..

# 3. 手动执行install（把devel里新生成的fusion复制到install）
catkin_make install -DCMAKE_BUILD_TYPE=Release -DBUILD_TESTING=OFF
```

```{code-block} shell
:linenos:
:caption: 编译obstacle_fusion
# 1. 先停进程，防止文件占用
pkill -f fusion || true
# 2. 清这个包的编译缓存
rm -rf build/target_tracking devel/target_tracking
# 3. 只编译fusion目标，这一步很快
cd build
make fusion -j$(nproc)
cd ..
# 4. 【重点替换】直接cp，代替catkin_make install
cp devel/lib/target_tracking/fusion install/lib/target_tracking/fusion
# 5. 刷新环境
source install/setup.bash
# 校验时间
ls -l install/lib/target_tracking/fusion
```
